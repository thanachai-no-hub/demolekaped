    create schema duck;   -- ตารางของเลขาเป็ดทั้งหมดอยู่ที่นี่

create table duck.line_links (
  employee_id  text primary key references public.employees(emp_id),
  line_user_id text unique not null,
  linked_at    timestamptz default now(),
  active       boolean default true
);

create view duck.profiles with (security_invoker = true) as
select e.emp_id    as id,
       e.full_name as display_name,
       e.dept_id   as department_id,
       case when e.position = 'CEO' then 'ceo'
            when e.is_manager      then 'head'
            else 'staff' end as role,
       l.line_user_id
  from public.employees e
  left join duck.line_links l on l.employee_id = e.emp_id and l.active
 where e.status = 'active';   -- คนลาออกใช้เลขาเป็ดไม่ได้ทันที

-- เลขาเป็ดได้สิทธิ์แค่ "อ่าน" ตารางพนักงาน
grant select on public.employees to duck_app;

    alter table tasks enable row level security;

create policy "เห็นงานตาม role" on tasks for select using (
  exists (select 1 from profiles me where me.id = auth.uid() and (
       me.role = 'ceo'
    or (me.role = 'head'  and me.department_id = tasks.department_id)
    or (me.role = 'staff' and tasks.owner_id = me.id)
  ))
);
